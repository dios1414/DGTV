<script>

/* =====================================================
   ÚNICA COSA QUE TENÉS QUE CAMBIAR
   ===================================================== */

const M3U_URL = "https://bit.ly/4ydeAZ8  ";


/* =====================================================
   VARIABLES
   ===================================================== */

let canales = [];
let indiceSeleccionado = 0;


/* =====================================================
   CARGAR M3U
   ===================================================== */

async function cargarM3U() {

  const contenido = document.getElementById("contenido");

  try {

    const respuesta = await fetch(M3U_URL, {
      cache: "no-store"
    });

    if (!respuesta.ok) {
      throw new Error("No se pudo descargar la lista");
    }

    const texto = await respuesta.text();

    canales = analizarM3U(texto);

    if (!canales.length) {
      contenido.innerHTML =
        '<div class="mensaje">No se encontraron canales.</div>';
      return;
    }

    mostrarCanales(canales);

  } catch (error) {

    console.error(error);

    contenido.innerHTML = `
      <div class="mensaje">
        No se pudo cargar la lista M3U.<br><br>
        Revisá la URL y que el servidor permita CORS.
      </div>
    `;
  }
}


/* =====================================================
   ANALIZAR M3U
   ===================================================== */

function analizarM3U(texto) {

  const lineas = texto
    .split(/\r?\n/)
    .map(x => x.trim())
    .filter(Boolean);

  const resultado = [];

  for (let i = 0; i < lineas.length; i++) {

    if (!lineas[i].startsWith("#EXTINF")) {
      continue;
    }

    const info = lineas[i];

    let url = "";

    for (let j = i + 1; j < lineas.length; j++) {

      if (
        lineas[j] &&
        !lineas[j].startsWith("#")
      ) {
        url = lineas[j];
        break;
      }
    }

    if (!url) continue;

    const nombreParte = info.substring(
      info.lastIndexOf(",") + 1
    ).trim();

    const grupoMatch = info.match(
      /group-title="([^"]*)"/i
    );

    const logoMatch = info.match(
      /tvg-logo="([^"]*)"/i
    );

    const nombreMatch = info.match(
      /tvg-name="([^"]*)"/i
    );

    resultado.push({
      nombre: nombreMatch?.[1] || nombreParte || "Canal",
      grupo: grupoMatch?.[1] || "Canales",
      logo: logoMatch?.[1] || "",
      url: url
    });
  }

  return resultado;
}


/* =====================================================
   MOSTRAR CANALES
   ===================================================== */

function mostrarCanales(lista) {

  const contenido = document.getElementById("contenido");

  contenido.innerHTML = "";

  const grupos = {};

  lista.forEach(canal => {

    if (!grupos[canal.grupo]) {
      grupos[canal.grupo] = [];
    }

    grupos[canal.grupo].push(canal);
  });


  Object.keys(grupos).forEach(grupo => {

    const categoria = document.createElement("section");
    categoria.className = "categoria";

    const titulo = document.createElement("div");
    titulo.className = "titulo-categoria";
    titulo.textContent = grupo;

    const canalesDiv = document.createElement("div");
    canalesDiv.className = "canales";


    grupos[grupo].forEach(canal => {

      const tarjeta = document.createElement("div");

      tarjeta.className = "canal";
      tarjeta.tabIndex = 0;


      if (canal.logo) {

        const img = document.createElement("img");

        img.src = canal.logo;
        img.alt = canal.nombre;

        img.onerror = function() {
          this.style.display = "none";
        };

        tarjeta.appendChild(img);
      }


      const nombre = document.createElement("div");

      nombre.className = "nombre";
      nombre.textContent = canal.nombre;

      tarjeta.appendChild(nombre);


      tarjeta.addEventListener("click", () => {
        reproducir(canal);
      });


      tarjeta.addEventListener("keydown", event => {

        if (
          event.key === "Enter" ||
          event.key === " "
        ) {
          event.preventDefault();
          reproducir(canal);
        }

      });


      canalesDiv.appendChild(tarjeta);

    });


    categoria.appendChild(titulo);
    categoria.appendChild(canalesDiv);

    contenido.appendChild(categoria);

  });


  activarNavegacionTV();
}


/* =====================================================
   REPRODUCTOR
   ===================================================== */

function reproducir(canal) {

  const reproductor =
    document.getElementById("reproductor");

  const video =
    document.getElementById("video");

  const titulo =
    document.getElementById("tituloVideo");


  titulo.textContent = canal.nombre;

  video.src = canal.url;

  reproductor.style.display = "block";

  video.play().catch(() => {});

}


/* =====================================================
   CERRAR REPRODUCTOR
   ===================================================== */

function cerrarReproductor() {

  const reproductor =
    document.getElementById("reproductor");

  const video =
    document.getElementById("video");

  video.pause();
  video.removeAttribute("src");
  video.load();

  reproductor.style.display = "none";
}


document
  .getElementById("cerrar")
  .addEventListener("click", cerrarReproductor);


/* =====================================================
   BOTÓN ATRÁS / ESC
   ===================================================== */

document.addEventListener("keydown", event => {

  if (event.key === "Escape" || event.key === "Backspace") {

    const reproductor =
      document.getElementById("reproductor");

    if (reproductor.style.display === "block") {

      cerrarReproductor();

    }

  }

});


/* =====================================================
   BUSCADOR
   ===================================================== */

document
  .getElementById("buscador")
  .addEventListener("input", function() {

    const texto =
      this.value.toLowerCase().trim();

    if (!texto) {

      mostrarCanales(canales);
      return;

    }

    const filtrados = canales.filter(canal =>

      canal.nombre
        .toLowerCase()
        .includes(texto) ||

      canal.grupo
        .toLowerCase()
        .includes(texto)

    );

    mostrarCanales(filtrados);

  });


/* =====================================================
   NAVEGACIÓN CON CONTROL
   ===================================================== */

function activarNavegacionTV() {

  const tarjetas =
    Array.from(document.querySelectorAll(".canal"));

  if (!tarjetas.length) return;

  tarjetas.forEach((tarjeta, indice) => {

    tarjeta.addEventListener("focus", () => {

      indiceSeleccionado = indice;

      tarjetas.forEach(t =>
        t.classList.remove("seleccionado")
      );

      tarjeta.classList.add("seleccionado");

    });

  });


  tarjetas[0].focus();
}


/* =====================================================
   INICIAR
   ===================================================== */

cargarM3U();
